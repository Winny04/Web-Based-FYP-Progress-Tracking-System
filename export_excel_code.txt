    [HttpGet]
    public IActionResult ExportUsersExcel(string type, string search)
    {
        var usersQuery = _context.ApplicationUsers.Where(u => u.UserType != "Admin" && u.UserType != "SuperAdmin");
        
        if (!string.IsNullOrEmpty(type))
        {
            usersQuery = usersQuery.Where(u => u.UserType == type);
        }
        
        if (!string.IsNullOrEmpty(search))
        {
            search = search.ToLower();
            usersQuery = usersQuery.Where(u => (u.FullName != null && u.FullName.ToLower().Contains(search))
                || (u.UserName != null && u.UserName.ToLower().Contains(search))
                || (u.Email != null && u.Email.ToLower().Contains(search)));
        }
        
        var users = usersQuery.ToList();
        var projects = _context.FYPProjects.Include(p => p.Supervisor).Include(p => p.Student).ToList();
        
        using (var workbook = new ClosedXML.Excel.XLWorkbook())
        {
            var worksheet = workbook.Worksheets.Add("Users Report");
            
            string filterText = "All Users";
            if (!string.IsNullOrEmpty(type)) filterText = type + "s";
            if (!string.IsNullOrEmpty(search)) filterText += $" (Search: '{search}')";
            
            worksheet.Cell(1, 1).Value = "User Management Report";
            worksheet.Cell(1, 1).Style.Font.Bold = true;
            worksheet.Cell(1, 1).Style.Font.FontSize = 16;
            worksheet.Range(1, 1, 1, 10).Merge();
            
            worksheet.Cell(2, 1).Value = $"Generated: {DateTime.Now.ToString("dd MMM yyyy, hh:mm tt")}";
            worksheet.Cell(2, 1).Style.Font.Italic = true;
            worksheet.Range(2, 1, 2, 10).Merge();
            
            worksheet.Cell(3, 1).Value = $"Filter: {filterText}";
            worksheet.Cell(3, 1).Style.Font.Italic = true;
            worksheet.Range(3, 1, 3, 10).Merge();
            
            int row = 5;
            worksheet.Cell(row, 1).Value = "ID";
            worksheet.Cell(row, 2).Value = "Full Name";
            worksheet.Cell(row, 3).Value = "Username / Student ID";
            worksheet.Cell(row, 4).Value = "Email";
            worksheet.Cell(row, 5).Value = "Role";
            worksheet.Cell(row, 6).Value = "Assignment";
            worksheet.Cell(row, 7).Value = "Setup";
            worksheet.Cell(row, 8).Value = "Created Date";
            worksheet.Cell(row, 9).Value = "Status";
            worksheet.Cell(row, 10).Value = "Login Status";
            
            int maxCols = 10;
            if (string.IsNullOrEmpty(type) || type == "Supervisor")
            {
                worksheet.Cell(row, 11).Value = "Assigned Count";
                worksheet.Cell(row, 12).Value = "Assigned Student Names";
                maxCols = 12;
            }
            
            var headerRow = worksheet.Range(row, 1, row, maxCols);
            headerRow.Style.Font.Bold = true;
            
            row++;
            
            foreach (var u in users)
            {
                worksheet.Cell(row, 1).Value = u.Id;
                worksheet.Cell(row, 2).Value = u.FullName ?? "-";
                worksheet.Cell(row, 3).Value = u.UserName ?? "-";
                worksheet.Cell(row, 4).Value = u.Email ?? "-";
                worksheet.Cell(row, 5).Value = u.UserType ?? "-";
                
                string assignment = "-";
                if (u.UserType == "Student")
                {
                    assignment = projects.FirstOrDefault(p => p.StudentId == u.Id)?.Supervisor?.FullName ?? "Not Assigned";
                }
                worksheet.Cell(row, 6).Value = assignment;
                
                bool isProfileIncomplete = ((u.UserType == "Supervisor" || u.UserType == "Examiner") && 
                                          (u.DepartmentId == null || 
                                           string.IsNullOrWhiteSpace(u.Programme) || 
                                           string.IsNullOrWhiteSpace(u.Specialization) || 
                                           string.IsNullOrWhiteSpace(u.Title)));
                
                worksheet.Cell(row, 7).Value = (u.UserType == "Student") ? "-" : (isProfileIncomplete ? "Pending" : "Completed");
                worksheet.Cell(row, 8).Value = u.DateCreated.ToString("dd MMM yyyy, hh:mm tt");
                worksheet.Cell(row, 9).Value = u.IsResign ? "Deactivated" : "Active";
                worksheet.Cell(row, 10).Value = u.RequirePasswordChange ? "First Login Pending" : "Has Logged In";
                
                if (maxCols == 12)
                {
                    if (u.UserType == "Supervisor")
                    {
                        int count = projects.Count(p => p.SupervisorId == u.Id && p.Student != null);
                        worksheet.Cell(row, 11).Value = count;
                        
                        var students = projects.Where(p => p.SupervisorId == u.Id && p.Student != null)
                                               .Select(p => p.Student.FullName)
                                               .ToList();
                        worksheet.Cell(row, 12).Value = students.Any() ? string.Join("; ", students) : "None";
                    }
                    else
                    {
                        worksheet.Cell(row, 11).Value = "-";
                        worksheet.Cell(row, 12).Value = "-";
                    }
                }
                
                row++;
            }
            
            if (users.Count > 0)
            {
                var dataRange = worksheet.Range(5, 1, row - 1, maxCols);
                var table = dataRange.CreateTable();
                table.Theme = ClosedXML.Excel.XLTableTheme.TableStyleMedium2;
                table.ShowAutoFilter = true;
            }
            
            worksheet.SheetView.FreezeRows(5);
            worksheet.Columns().AdjustToContents();
            
            // Cap widths and wrap text
            if (worksheet.Column(4).Width > 40) worksheet.Column(4).Width = 40;
            worksheet.Column(4).Style.Alignment.WrapText = true;
            
            if (maxCols == 12)
            {
                if (worksheet.Column(12).Width > 50) worksheet.Column(12).Width = 50;
                worksheet.Column(12).Style.Alignment.WrapText = true;
            }
            
            // Vertical alignment
            worksheet.Style.Alignment.Vertical = ClosedXML.Excel.XLAlignmentVerticalValues.Top;
            
            using (var stream = new System.IO.MemoryStream())
            {
                workbook.SaveAs(stream);
                var content = stream.ToArray();
                return File(content, "application/vnd.openxmlformats-officedocument.spreadsheetml.sheet", "User_Management_Report.xlsx");
            }
        }
    }
