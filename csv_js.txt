        // --- NEW CSV UPLOAD LOGIC ---
        var csvValidRows = [];
        var csvAllRows = [];

        $('#csvUserType').on('change', function () {
            var type = $(this).val();
            if (type) {
                $('#csvFormatInfo, #csvUploadArea').slideDown();
                var downloadUrl = '/Admin/DownloadCSVTemplate?userType=' + type;
                $('#csvDownloadTemplateBtn').attr('href', downloadUrl);

                if (type === 'Student') {
                    $('#csvFormatDetails').html('<strong>Columns:</strong> <code>Full Name, Username, Programme, Academic Session</code><br/>Username must be <code>p</code> followed by 8 digits (e.g. <code>p22014692</code>).');
                } else if (type === 'Supervisor') {
                    $('#csvFormatDetails').html('<strong>Columns:</strong> <code>Full Name, Username, Department, Title, Max Student Quota</code>');
                } else if (type === 'Examiner') {
                    $('#csvFormatDetails').html('<strong>Columns:</strong> <code>Full Name, Username, Department, Title</code>');
                }
            } else {
                $('#csvFormatInfo, #csvUploadArea, #csvValidateBtn').slideUp();
                $('#csvFileInput').val('');
                $('#csvFileInfo').addClass('d-none');
            }
        });

        $('#csvDropzone').on('click', function () {
            $('#csvFileInput').click();
        });
        $('#csvDropzone').on('dragover', function (e) {
            e.preventDefault();
            $(this).addClass('bg-light-primary border-primary');
        }).on('dragleave drop', function (e) {
            e.preventDefault();
            $(this).removeClass('bg-light-primary border-primary');
        });
        $('#csvDropzone').on('drop', function (e) {
            var files = e.originalEvent.dataTransfer.files;
            if (files.length > 0) {
                $('#csvFileInput')[0].files = files;
                handleCsvFileSelect();
            }
        });
        $('#csvFileInput').on('change', handleCsvFileSelect);

        function handleCsvFileSelect() {
            var file = $('#csvFileInput')[0].files[0];
            if (file) {
                if (file.name.toLowerCase().endsWith('.csv')) {
                    $('#csvFileName').text(file.name);
                    $('#csvFileSize').text((file.size / 1024).toFixed(1) + ' KB');
                    $('#csvFileInfo').removeClass('d-none');
                    $('#csvValidateBtn').removeClass('d-none').hide().slideDown();
                } else {
                    Swal.fire('Error', 'Please select a valid .csv file', 'error');
                    $('#csvFileInput').val('');
                    $('#csvFileInfo').addClass('d-none');
                    $('#csvValidateBtn').slideUp();
                }
            }
        }

        $('#csvValidateBtn').on('click', function () {
            var file = $('#csvFileInput')[0].files[0];
            var type = $('#csvUserType').val();
            if (!file || !type) return;

            var formData = new FormData();
            formData.append('csvFile', file);
            formData.append('csvUserType', type);

            var btn = $(this);
            var originalText = btn.html();
            btn.prop('disabled', true).html('<span class="spinner-border spinner-border-sm me-2"></span>Validating...');

            $.ajax({
                url: '/Admin/ValidateCSV',
                type: 'POST',
                data: formData,
                processData: false,
                contentType: false,
                success: function (res) {
                    btn.prop('disabled', false).html(originalText);
                    if (res.success) {
                        csvAllRows = res.rows;
                        csvValidRows = res.rows.filter(r => r.isValid);
                        
                        var total = csvAllRows.length;
                        var valid = csvValidRows.length;
                        var invalid = total - valid;

                        $('#csvValidationSummary').html('<strong>' + total + ' rows detected.</strong> ' + valid + ' ready to import. ' + invalid + ' rows invalid or skipped.');
                        
                        var tbody = $('#csvPreviewBody');
                        tbody.empty();
                        
                        csvAllRows.forEach(function (r) {
                            var statusBadge = r.isValid 
                                ? '<span class="badge badge-light-success fs-8 px-2">Ready</span>'
                                : '<span class="badge badge-light-danger fs-8 px-2">' + r.status + '</span>';
                                
                            var emailDisplay = r.generatedEmail ? '<a href="mailto:' + r.generatedEmail + '" class="text-dark text-hover-primary">' + r.generatedEmail + '</a>' : '-';

                            tbody.append(
                                '<tr>' +
                                    '<td>' + r.rowIndex + '</td>' +
                                    '<td>' + (r.fullName || '-') + '</td>' +
                                    '<td>' + (r.username || '-') + '</td>' +
                                    '<td>' + emailDisplay + '</td>' +
                                    '<td>' + statusBadge + '</td>' +
                                '</tr>'
                            );
                        });

                        if (invalid > 0) {
                            $('#csvDownloadErrorsBtn').removeClass('d-none');
                        } else {
                            $('#csvDownloadErrorsBtn').addClass('d-none');
                        }

                        if (valid > 0) {
                            $('#csvConfirmImportBtn').prop('disabled', false).text('Import ' + valid + ' Valid Users');
                        } else {
                            $('#csvConfirmImportBtn').prop('disabled', true).text('Nothing to Import');
                        }

                        $('#csvStep1').hide();
                        $('#csvStep2').fadeIn();
                    } else {
                        Swal.fire('Error', res.message, 'error');
                    }
                },
                error: function () {
                    btn.prop('disabled', false).html(originalText);
                    Swal.fire('Error', 'An error occurred during validation.', 'error');
                }
            });
        });

        $('#csvBackBtn').on('click', function () {
            $('#csvStep2').hide();
            $('#csvStep1').fadeIn();
        });

        $('#csvDownloadErrorsBtn').on('click', function () {
            var invalidRows = csvAllRows.filter(r => !r.isValid);
            var csvContent = 'Row,Original Content,Error Reason\n';
            invalidRows.forEach(function (r) {
                var rowStr = r.originalRow.replace(/"/g, '""');
                csvContent += r.rowIndex + ',"' + rowStr + '","' + r.status + '"\n';
            });
            var blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
            var url = URL.createObjectURL(blob);
            var a = document.createElement('a');
            a.href = url;
            a.download = 'import_errors.csv';
            document.body.appendChild(a);
            a.click();
            document.body.removeChild(a);
        });

        $('#csvConfirmImportBtn').on('click', function () {
            var btn = $(this);
            var type = $('#csvUserType').val();
            
            btn.prop('disabled', true).html('<span class="spinner-border spinner-border-sm me-2"></span>Importing...');
            
            $.ajax({
                url: '/Admin/ConfirmImportCSV',
                type: 'POST',
                contentType: 'application/json',
                data: JSON.stringify({
                    userType: type,
                    rows: csvAllRows // sending all to keep row numbers, backend filters valid ones
                }),
                success: function (res) {
                    if (res.success) {
                        $('#csvUploadModal').modal('hide');
                        Swal.fire('Success!', res.message, 'success').then(() => {
                            location.reload();
                        });
                    } else {
                        btn.prop('disabled', false).text('Confirm Import');
                        Swal.fire('Error', res.message, 'error');
                    }
                },
                error: function () {
                    btn.prop('disabled', false).text('Confirm Import');
                    Swal.fire('Error', 'An error occurred during import.', 'error');
                }
            });
        });
        // --- END NEW CSV UPLOAD LOGIC ---
