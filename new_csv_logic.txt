        // -- CSV Upload Workflow --
        var validatedRowsData = null;

        $('#csvUserType').on('change', function () {
            var type = $(this).val();
            if (type) {
                $('#csvUploadArea').slideDown();
                $('#csvFormatInfo').slideDown();
                
                var formatHtml = '<strong>Required Columns:</strong><br/>';
                if (type === 'Student') {
                    formatHtml += 'Full Name, Username, Programme, Academic Session';
                } else if (type === 'Supervisor') {
                    formatHtml += 'Full Name, Username, Department, Title, Max Student Quota';
                } else if (type === 'Examiner') {
                    formatHtml += 'Full Name, Username, Department, Title';
                }
                $('#csvFormatDetails').html(formatHtml);
                
                // Set template download link based on type
                $('#csvDownloadTemplateBtn').attr('href', '/Admin/DownloadCSVTemplate?type=' + type);
            } else {
                $('#csvUploadArea').slideUp();
                $('#csvFormatInfo').slideUp();
            }
        });

        $('#csvDropzone').on('click', function() {
            $('#csvFileInput').click();
        });

        $('#csvFileInput').on('change', function() {
            var file = this.files[0];
            if (file) {
                $('#csvFileName').text(file.name);
                var size = (file.size / 1024).toFixed(2) + ' KB';
                $('#csvFileSize').text(size);
                $('#csvFileInfo').removeClass('d-none');
                $('#csvValidateBtn').removeClass('d-none');
            }
        });

        $('#csvValidateBtn').on('click', function() {
            var file = $('#csvFileInput')[0].files[0];
            var type = $('#csvUserType').val();
            if (!file || !type) return;

            var formData = new FormData();
            formData.append('csvFile', file);
            formData.append('csvUserType', type);

            var btn = $(this);
            btn.prop('disabled', true).html('<i class="fas fa-spinner fa-spin me-2"></i>Validating...');

            $.ajax({
                url: '/Admin/ValidateCSV',
                type: 'POST',
                data: formData,
                processData: false,
                contentType: false,
                success: function(response) {
                    btn.prop('disabled', false).html('<i class="fas fa-check-circle me-2"></i>Validate CSV');
                    
                    if (response.success) {
                        validatedRowsData = response.data;
                        
                        var validCount = response.data.rows.filter(r => r.isValid).length;
                        var errorCount = response.data.rows.filter(r => !r.isValid).length;
                        
                        var summaryHtml = `<strong>Validation Complete!</strong><br/>Found ${validCount} valid users and ${errorCount} errors.`;
                        $('#csvValidationSummary').html(summaryHtml).removeClass('alert-light-warning alert-light-danger').addClass(errorCount > 0 ? 'alert-light-warning' : 'alert-light-success');
                        
                        var tbody = $('#csvPreviewBody');
                        tbody.empty();
                        
                        response.data.rows.forEach(function(row) {
                            var trClass = row.isValid ? '' : 'bg-light-danger';
                            var statusHtml = row.isValid 
                                ? '<span class="badge badge-light-success">Valid</span>' 
                                : `<span class="badge badge-light-danger" data-bs-toggle="tooltip" title="${row.status}">Error</span>`;
                            
                            var tr = `<tr class="${trClass}">
                                <td>${row.rowIndex}</td>
                                <td>${row.fullName || '-'}</td>
                                <td>${row.username || '-'}</td>
                                <td>${row.email || '-'}</td>
                                <td>${statusHtml}</td>
                            </tr>`;
                            tbody.append(tr);
                        });
                        
                        if (errorCount > 0) {
                            $('#csvDownloadErrorsBtn').removeClass('d-none');
                        } else {
                            $('#csvDownloadErrorsBtn').addClass('d-none');
                        }
                        
                        $('#csvConfirmImportBtn').prop('disabled', validCount === 0);

                        $('[data-bs-toggle="tooltip"]').tooltip();

                        $('#csvStep1').hide();
                        $('#csvStep2').fadeIn();
                    } else {
                        Swal.fire('Error', response.message, 'error');
                    }
                },
                error: function() {
                    btn.prop('disabled', false).html('<i class="fas fa-check-circle me-2"></i>Validate CSV');
                    Swal.fire('Error', 'Failed to communicate with server.', 'error');
                }
            });
        });

        $('#csvBackBtn').on('click', function() {
            $('#csvStep2').hide();
            $('#csvStep1').fadeIn();
        });

        $('#csvConfirmImportBtn').on('click', function() {
            if (!validatedRowsData) return;
            
            var btn = $(this);
            btn.prop('disabled', true).html('<i class="fas fa-spinner fa-spin me-2"></i>Importing...');

            $.ajax({
                url: '/Admin/ConfirmImportCSV',
                type: 'POST',
                contentType: 'application/json',
                data: JSON.stringify(validatedRowsData),
                success: function(response) {
                    btn.prop('disabled', false).html('Confirm Import');
                    if (response.success) {
                        $('#csvUploadModal').modal('hide');
                        Swal.fire('Success', response.message, 'success').then(() => {
                            window.location.reload();
                        });
                    } else {
                        Swal.fire('Error', response.message, 'error');
                    }
                },
                error: function() {
                    btn.prop('disabled', false).html('Confirm Import');
                    Swal.fire('Error', 'Failed to import users.', 'error');
                }
            });
        });
